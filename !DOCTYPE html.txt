<script>

const message = `Dear Proma,

Maybe this is just a small webpage,
but the feelings behind it are real.

In a world full of ordinary moments,
you are one of the rare things that
makes everything feel a little more beautiful.

Your smile,
your presence,
even the thought of you
can make a normal day feel special.

So I made this little letter for you,
something simple,
but something created only for you.

Because some people deserve
more than just a message.

They deserve something memorable.

And to me,
you are unforgettable. ❤️
`;

let i=0;

function typeWriter(){
if(i<message.length){
document.getElementById("text").innerHTML+=message.charAt(i);
i++;
setTimeout(typeWriter,40);
}
}

function openLetter(){

document.querySelector(".envelope").style.display="none";

document.getElementById("letter").style.display="block";

document.getElementById("music").play();

typeWriter();

}

function hearts(){

heart.style.fontSize=(Math.random()*20+10)+"px";

document.body.appendChild(heart);

setTimeout(()=>{
heart.remove();
},6000);

}

setInterval(hearts,500);

</script>
